cmake_minimum_required(VERSION 3.16)
project(rm27_frozen_map_comparison)
set(CMAKE_CXX_STANDARD 17)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io)
find_package(nlohmann_json REQUIRED)
find_package(yaml-cpp REQUIRED)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(FROZEN_CORE "${REPO_ROOT}/tools/global_relocalization/.build/ros/install/global_relocalization/lib/libglobal_relocalization_core.so")
add_executable(offline_map_compare ../offline_map_compare.cpp)
target_include_directories(offline_map_compare PRIVATE "${REPO_ROOT}/src/global_relocalization/include" ${PCL_INCLUDE_DIRS})
target_link_libraries(offline_map_compare PRIVATE "${FROZEN_CORE}" Eigen3::Eigen ${PCL_LIBRARIES} nlohmann_json::nlohmann_json yaml-cpp)
