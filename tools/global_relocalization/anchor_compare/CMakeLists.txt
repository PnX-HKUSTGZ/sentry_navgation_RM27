cmake_minimum_required(VERSION 3.16)
project(relocalization_anchor_compare)
set(CMAKE_CXX_STANDARD 17)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io)
find_package(nlohmann_json REQUIRED)
find_package(yaml-cpp REQUIRED)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(anchor_compare anchor_compare.cpp)
target_include_directories(anchor_compare PRIVATE "${REPO_ROOT}/src/global_relocalization/include" ${PCL_INCLUDE_DIRS})
target_link_libraries(anchor_compare PRIVATE "${REPO_ROOT}/tools/global_relocalization/.build/ros/install/global_relocalization/lib/libglobal_relocalization_core.so" Eigen3::Eigen ${PCL_LIBRARIES} nlohmann_json::nlohmann_json yaml-cpp)
