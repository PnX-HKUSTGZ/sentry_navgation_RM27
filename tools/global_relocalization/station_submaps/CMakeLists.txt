cmake_minimum_required(VERSION 3.16)
project(sensor_station_submaps)
set(CMAKE_CXX_STANDARD 17)
find_package(Eigen3 REQUIRED)
find_package(OpenMP REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters kdtree)
find_package(nlohmann_json REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(ORIGINAL_GICP "${ROOT}/build/relocalization_manager/_deps/small_gicp-src/include")
set(GICP_LIBRARY "${ROOT}/tools/global_relocalization/.build/ros/build/global_relocalization/_deps/small_gicp-build/libsmall_gicp.so")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_frontend.py" "${ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/generated" COMMAND_ERROR_IS_FATAL ANY)
add_library(experimental_station_frontend "${CMAKE_CURRENT_BINARY_DIR}/generated/experimental_frontend.cpp")
target_include_directories(experimental_station_frontend PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/generated" "${ROOT}/src/global_relocalization/include" "${ORIGINAL_GICP}" ${PCL_INCLUDE_DIRS})
target_link_libraries(experimental_station_frontend PUBLIC Eigen3::Eigen OpenMP::OpenMP_CXX ${PCL_LIBRARIES} ${GICP_LIBRARY} nlohmann_json::nlohmann_json)
add_executable(offline_station_submaps offline_station_submaps.cpp)
target_link_libraries(offline_station_submaps experimental_station_frontend)
enable_testing()
add_test(NAME trusted_calibration_contract COMMAND offline_station_submaps --self-test)
