cmake_minimum_required(VERSION 3.16)
project(pointlio_sensor_tests)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
find_package(Eigen3 REQUIRED)
set(SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../.build/mapping_source")
add_executable(test_instantaneous_preprocess test_instantaneous_preprocess.cpp "${SOURCE}/src/preprocess.cpp")
target_include_directories(test_instantaneous_preprocess PRIVATE "${SOURCE}/src")
ament_target_dependencies(test_instantaneous_preprocess rclcpp pcl_conversions livox_ros_driver2)
add_executable(test_measurement_variance test_measurement_variance.cpp)
target_include_directories(test_measurement_variance PRIVATE "${SOURCE}/include")
target_link_libraries(test_measurement_variance Eigen3::Eigen)
enable_testing()
add_test(NAME instantaneous_preprocess COMMAND test_instantaneous_preprocess)
add_test(NAME measurement_variance COMMAND test_measurement_variance)
add_test(NAME converter_timing COMMAND /usr/bin/python3
  "${CMAKE_CURRENT_SOURCE_DIR}/test_converter_timing.py" --converter
  "${CMAKE_CURRENT_SOURCE_DIR}/../.build/sensor_timing_ros/install/ign_sim_pointcloud_tool/lib/ign_sim_pointcloud_tool/ign_sim_pointcloud_tool_node")
set_tests_properties(converter_timing PROPERTIES ENVIRONMENT
  "ROS_DOMAIN_ID=83;LD_LIBRARY_PATH=${CMAKE_CURRENT_SOURCE_DIR}/../.build/sensor_timing_ros/install/ign_sim_pointcloud_tool/lib:$ENV{LD_LIBRARY_PATH}")
add_test(NAME imu_queue_depth COMMAND /usr/bin/python3
  "${CMAKE_CURRENT_SOURCE_DIR}/test_imu_queue_depth.py")
set_tests_properties(imu_queue_depth PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=84")
add_executable(test_loop_clock test_loop_clock.cpp)
ament_target_dependencies(test_loop_clock rclcpp)
add_test(NAME loop_clock COMMAND test_loop_clock)
set_tests_properties(loop_clock PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=84")
