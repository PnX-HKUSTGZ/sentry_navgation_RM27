cmake_minimum_required(VERSION 3.16)
project(raw_gicp_mapping)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io)
find_package(nlohmann_json REQUIRED)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(FROZEN_CORE "${REPO_ROOT}/tools/global_relocalization/.build/ros/install/global_relocalization/lib/libglobal_relocalization_core.so")
if(NOT EXISTS "${FROZEN_CORE}")
  message(FATAL_ERROR "Build the existing global_relocalization core first")
endif()
add_executable(raw_mapping_node raw_mapping_node.cpp)
target_include_directories(raw_mapping_node PRIVATE "${REPO_ROOT}/src/global_relocalization/include" ${PCL_INCLUDE_DIRS})
target_link_libraries(raw_mapping_node "${FROZEN_CORE}" Eigen3::Eigen ${PCL_LIBRARIES} nlohmann_json::nlohmann_json)
ament_target_dependencies(raw_mapping_node rclcpp sensor_msgs nav_msgs std_msgs pcl_conversions)
enable_testing()
find_package(ament_cmake_gtest REQUIRED)
ament_add_gtest(test_mapping_output test_mapping_output.cpp)
target_include_directories(test_mapping_output PRIVATE "${REPO_ROOT}/src/global_relocalization/include" ${PCL_INCLUDE_DIRS})
target_link_libraries(test_mapping_output "${FROZEN_CORE}" Eigen3::Eigen ${PCL_LIBRARIES})
ament_target_dependencies(test_mapping_output rclcpp sensor_msgs nav_msgs pcl_conversions)
