cmake_minimum_required(VERSION 3.16)
project(rm27_slam_graph_export)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(slam_toolbox REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(Boost REQUIRED COMPONENTS serialization filesystem)
find_library(KARTO_LIBRARY kartoSlamToolbox HINTS /opt/ros/jazzy/lib REQUIRED)
add_executable(slam_graph_export ../slam_graph_export.cpp)
ament_target_dependencies(slam_graph_export rclcpp rclcpp_lifecycle slam_toolbox)
target_link_libraries(slam_graph_export ${KARTO_LIBRARY} Eigen3::Eigen Boost::serialization Boost::filesystem)
